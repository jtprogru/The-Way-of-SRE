import Spider from '../../components/Spider.astro';
import { countLeaves, roadmap } from '../../data/roadmap.ts';
import { navFor } from '../../data/nav.ts';

export const base = import.meta.env.BASE_URL.replace(/\/$/, '');
// Строка «Глубже — …» остаётся прозой, но состав ссылок берётся из nav.ts.
export const introNav = navFor('intro');
// countLeaves считает и подлисты: для читателя вложенная практика — такая
// же написанная страница, как и любая другая.
export const totalLeaves = roadmap.branches.reduce(
  (sum, b) => sum + b.l1.reduce((s, l1) => s + countLeaves(l1), 0),
  0,
);
export const totalL1 = roadmap.branches.reduce((sum, b) => sum + b.l1.length, 0);
// Числа берутся из данных, поэтому и окончание считается, а не пишется
// руками: «22 компетенций» и «23 компетенции» — разные формы.
export const plural = (n, one, few, many) => {
  const mod10 = n % 10;
  const mod100 = n % 100;
  if (mod10 === 1 && mod100 !== 11) return one;
  if (mod10 >= 2 && mod10 <= 4 && (mod100 < 12 || mod100 > 14)) return few;
  return many;
};
