import MindMap from '../../components/MindMap.astro';
import { countLeaves, roadmap } from '../../data/roadmap.ts';

export const totalL1 = roadmap.branches.reduce((sum, b) => sum + b.l1.length, 0);
export const totalLeaves = roadmap.branches.reduce(
  (sum, b) => sum + b.l1.reduce((s, l1) => s + countLeaves(l1), 0),
  0,
);

Здесь весь роадмап одним холстом: {roadmap.branches.length} ветви, {totalL1} компетенции верхнего уровня и {totalLeaves} листьев с подлистами. Карта растёт в обе стороны от корня — SRE Engineering справа, SRE Culture и SRE Practices слева. Кликается каждая плашка на любом уровне.

Карта — витрина, а не рабочий навигатор. Чтобы понять, с чего начинать, смотрите [роадмап по приоритетам](/The-Way-of-SRE/priorities/); чтобы двигаться по материалу — левый сайдбар и [карту компетенций на главной](/The-Way-of-SRE/). Здесь же видно другое: сколько всего входит в роль и как одно связано с другим.

Цвет плашки — ветвь, точка на плашке — приоритет компетенции или практики. Пунктирная рамка означает подлист: практика, уточняющая соседнюю, а не самостоятельная тема. Состав карты собирается из тех же данных, что сайдбар и страницы ветвей, поэтому новый лист появляется здесь сам.
